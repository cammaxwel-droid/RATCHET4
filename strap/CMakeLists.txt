cmake_minimum_required(VERSION 3.16)
project(StrapReview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/include/ratchet4.hpp" RATCHET_SHA)
if(NOT RATCHET_SHA STREQUAL "afc73d26c5c7b33448f6e11a33c081c3d719a70183b70801da3d5f32f81ef1ae")
  message(FATAL_ERROR "Strap requires the supplied original Ratchet-4 V1 header")
endif()
add_library(strap INTERFACE)
target_include_directories(strap INTERFACE include)
add_executable(strap_demo src/demo.cpp)
target_link_libraries(strap_demo PRIVATE strap)
add_executable(test_strap tests/test_strap.cpp)
target_link_libraries(test_strap PRIVATE strap)
enable_testing()
add_test(NAME strap_contract COMMAND test_strap)
add_executable(strap_benchmark tests/benchmark.cpp)
target_link_libraries(strap_benchmark PRIVATE strap)
